    <script>
        // ⚠️ ВСТАВЬ СВОИ ДАННЫЕ ОТ TWITCH ⚠️
        const CLIENT_ID = 'ТВОЙ_CLIENT_ID'; // Например: 'abc123def456'
        const ACCESS_TOKEN = 'ТВОЙ_ACCESS_TOKEN'; // Получить тут: https://twitchtokengenerator.com/

        // 🔥 СПИСОК ТВОИХ СТРИМЕРОВ (пиши их логины)
        const STREAMERS = [
            'scarleycrystal',
            'trewoga',
            'bredinaplay',
            'nitablade',
            'morgimori',
            'mrdentl'
        ];

        async function loadStreamers() {
            const list = document.getElementById('streamerList');
            list.innerHTML = '<p style="color:white;">Загрузка...</p>';

            try {
                // 1. Получаем ID стримеров по логинам
                const userRes = await fetch(`https://api.twitch.tv/helix/users?login=${STREAMERS.join('&login=')}`, {
                    headers: {
                        'Client-ID': CLIENT_ID,
                        'Authorization': `Bearer ${ACCESS_TOKEN}`
                    }
                });
                const userData = await userRes.json();
                const userIds = userData.data.map(u => u.id);

                // 2. Получаем информацию о стримах (кто в онлайне)
                const streamRes = await fetch(`https://api.twitch.tv/helix/streams?user_id=${userIds.join('&user_id=')}`, {
                    headers: {
                        'Client-ID': CLIENT_ID,
                        'Authorization': `Bearer ${ACCESS_TOKEN}`
                    }
                });
                const streamData = await streamRes.json();

                // 3. Сортируем по зрителям (как на скрине)
                const online = streamData.data.sort((a, b) => b.viewer_count - a.viewer_count);

                if (online.length === 0) {
                    list.innerHTML = '<p style="color:#adadb8;">Сейчас никто не стримит из списка 😴</p>';
                    return;
                }

                // 4. Рендерим карточки
                let html = '';
                online.forEach((stream, index) => {
                    const started = new Date(stream.started_at);
                    const now = new Date();
                    const diffMs = now - started;
                    const diffHrs = Math.floor(diffMs / (1000 * 60 * 60));
                    const diffMins = Math.floor((diffMs % (1000 * 60 * 60)) / (1000 * 60));
                    const timeStr = `${diffHrs} ч ${diffMins} мин`;

                    // Аватарка (используем фото пользователя)
                    const user = userData.data.find(u => u.id === stream.user_id);
                    const avatarUrl = user ? user.profile_image_url : '';

                    html += `
                        <div class="streamer-card" onclick="window.open('https://twitch.tv/${stream.user_login}', '_blank')">
                            <div class="rank">${index + 1}</div>
                            <div class="avatar"><img src="${avatarUrl}" alt="${stream.user_login}"></div>
                            <div class="info">
                                <div class="name">${stream.user_login} <small>${stream.user_name}</small></div>
                                <div class="game">🎮 ${stream.game_name}</div>
                            </div>
                            <div class="stats">
                                <div class="viewers">👁️ ${stream.viewer_count}</div>
                                <div class="live-time">🟣 идет ${timeStr}</div>
                            </div>
                        </div>
                    `;
                });

                list.innerHTML = html;

            } catch (error) {
                list.innerHTML = `<p style="color:red;">Ошибка загрузки: ${error.message}. Проверь CLIENT_ID и TOKEN.</p>`;
                console.error(error);
            }
        }

        // Запускаем
        loadStreamers();
        // Обновляем каждые 2 минуты (чтобы данные были свежие)
        setInterval(loadStreamers, 120000);
    </script>
